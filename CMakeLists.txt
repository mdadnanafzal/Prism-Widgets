# Alternative build (Visual Studio / MinGW on Windows). The tested build is build.sh (MinGW-w64 cross-compile).
cmake_minimum_required(VERSION 3.16)
project(PrismWidgets CXX RC)
set(CMAKE_CXX_STANDARD 17)
add_executable(PrismWidgets WIN32 src/app.cpp src/arrange.cpp src/settings.cpp src/widgets.cpp src/widgets2.cpp src/media.cpp src/weather.cpp src/events.cpp src/daypop.cpp src/net.cpp src/update.cpp src/peek.cpp src/reminders.cpp src/glass.cpp src/util.cpp src/json.cpp src/pet/pet_rig.cpp src/pet/pet_draw.cpp src/pet/pet_scenes.cpp src/pet/pet_brain.cpp src/pet/pet_input.cpp src/pet/pet_fx.cpp res/app.rc)
target_compile_definitions(PrismWidgets PRIVATE UNICODE _UNICODE)
target_link_libraries(PrismWidgets d2d1 dwrite windowscodecs ole32 uuid shell32 shlwapi shcore dwmapi wtsapi32 psapi runtimeobject winhttp bcrypt iphlpapi winmm advapi32 gdi32 user32)
if(MSVC)
  target_link_options(PrismWidgets PRIVATE /MANIFEST:NO /ENTRY:wWinMainCRTStartup)
  set_property(TARGET PrismWidgets PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
else()
  target_link_options(PrismWidgets PRIVATE -municode -static -s)
endif()
